        // Lógica de filtrado
        function filterEvents() {
            const selected = select.value;
            tbody.innerHTML = '';
            
            const filtered = selected === 'all' 
                ? events 
                : events.filter(e => e.mes === selected);
                
            if (filtered.length === 0) {
                tbody.innerHTML = '<tr><td colspan="2" style="text-align:center;">No hay eventos</td></tr>';
                return;
            }

            filtered.forEach(e => {
                const tr = document.createElement('tr');
                tr.innerHTML = `<td><strong>${e.fecha}</strong></td><td>${e.evento}</td>`;
